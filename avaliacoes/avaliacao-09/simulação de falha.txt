import 'dart:convert';
import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;

// 1. Modelo de Dados para mapear o JSON da API
class Usuario {
  final int id;
  final String nome;
  final String email;

  Usuario({required this.id, required this.nome, required this.email});

  factory Usuario.fromJson(Map<String, dynamic> json) {
    return Usuario(id: json['id'], nome: json['name'], email: json['email']);
  }
}

// 2. FUNÇÃO MODIFICADA APENAS PARA FORÇAR E SIMULAR O ERRO
Future<List<Usuario>> fetchUsuarios() async {
  // Força o aplicativo a lançar uma exceção imediatamente
  throw Exception('Falha na conexão com o servidor. Código: 500');
}

// 3. Inicialização do App e Interface Visual
void main() => runApp(MaterialApp(home: TelaUsuarios(), debugShowCheckedModeBanner: false));

class TelaUsuarios extends StatefulWidget {
  @override
  _TelaUsuariosState createState() => _TelaUsuariosState();
}

class _TelaUsuariosState extends State<TelaUsuarios> {
  late Future<List<Usuario>> _futureUsuarios;

  @override
  void initState() {
    super.initState();
    _futureUsuarios = fetchUsuarios(); // Dispara a busca ao iniciar
  }

  Future<void> _recarregarLista() async {
    setState(() { _futureUsuarios = fetchUsuarios(); });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('Usuários da API'), 
        backgroundColor: Colors.blue,
        foregroundColor: Colors.white,
      ),
      // RefreshIndicator adiciona o recurso de "puxar para baixo para atualizar"
      body: RefreshIndicator(
        onRefresh: _recarregarLista,
        child: FutureBuilder<List<Usuario>>(
          future: _futureUsuarios,
          builder: (context, snapshot) {
            // Se estiver carregando: mostra a rodinha de progresso
            if (snapshot.connectionState == ConnectionState.waiting) {
              return Center(child: CircularProgressIndicator());
            } 
            // Se der erro de rede ou servidor: mostra o erro em vermelho
            else if (snapshot.hasError) {
              return Center(
                child: Padding(
                  padding: const EdgeInsets.all(16.0),
                  child: Text('${snapshot.error}', style: TextStyle(color: Colors.red, fontSize: 16)),
                ),
              );
            } 
            // Se os dados chegarem com sucesso: monta a lista rolável
            else if (snapshot.hasData) {
              return ListView.builder(
                itemCount: snapshot.data!.length,
                itemBuilder: (context, index) {
                  Usuario usuario = snapshot.data![index];
                  return Card(
                    margin: EdgeInsets.symmetric(horizontal: 12, vertical: 6),
                    elevation: 2,
                    child: ListTile(
                      leading: CircleAvatar(
                        backgroundColor: Colors.blueAccent,
                        child: Text(usuario.id.toString(), style: TextStyle(color: Colors.white)),
                      ),
                      title: Text(usuario.nome, style: TextStyle(fontWeight: FontWeight.bold)),
                      subtitle: Text(usuario.email),
                    ),
                  );
                },
              );
            }
            return Center(child: Text('Nenhum dado encontrado.'));
          },
        ),
      ),
    );
  }
}
